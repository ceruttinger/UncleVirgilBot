timeline <- metadata %>%
  filter(include_in_timeline == TRUE) %>%
  left_join(stats %>% select(chapter, word_count), by = "chapter") %>%
  left_join(sentiment %>% select(chapter, net_bing_sentiment, afinn_score), by = "chapter") %>%
  mutate(midpoint_year = (as.numeric(approx_start_year) + as.numeric(approx_end_year)) / 2)

timeline %>%
  ggplot(aes(x = midpoint_year, y = net_bing_sentiment, label = chapter)) +
  geom_line() +
  geom_point() +
  geom_text(nudge_y = 5, check_overlap = TRUE) +
  labs(x = "Approximate midpoint year", y = "Net Bing sentiment", title = "Draft sentiment timeline") +
  theme_minimal()
