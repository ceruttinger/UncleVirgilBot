## Highest-weight chapter/entity links

```{r}
#| echo: false
#| message: false
#| warning: false

library(readr)
library(dplyr)
library(knitr)

edges <- read_csv("data/public/graph_edges.csv", show_col_types = FALSE)
nodes <- read_csv("data/public/graph_nodes.csv", show_col_types = FALSE)

edge_from <- intersect(c("from", "source", "source_id"), names(edges))[1]
edge_to <- intersect(c("to", "target", "target_id"), names(edges))[1]
edge_weight <- intersect(c("weight", "mentions", "value"), names(edges))[1]
node_id <- intersect(c("id", "node_id"), names(nodes))[1]
node_label <- intersect(c("label", "canonical_entity", "name"), names(nodes))[1]
node_type <- intersect(c("node_type", "entity_type", "group", "type"), names(nodes))[1]

if (!is.na(edge_from) && !is.na(edge_to) && !is.na(node_id) && !is.na(node_label)) {
  edges2 <- edges |>
    rename(from_id = all_of(edge_from), to_id = all_of(edge_to))

  if (!is.na(edge_weight)) {
    edges2 <- edges2 |> rename(mentions = all_of(edge_weight))
  } else {
    edges2 <- edges2 |> mutate(mentions = 1)
  }

  nodes2 <- nodes |>
    rename(graph_id = all_of(node_id), graph_label = all_of(node_label))

  if (!is.na(node_type)) {
    nodes2 <- nodes2 |> rename(graph_type = all_of(node_type))
  } else {
    nodes2 <- nodes2 |> mutate(graph_type = NA_character_)
  }

  top_links <- edges2 |>
    filter(grepl("chapter", from_id, ignore.case = TRUE) | grepl("chapter", to_id, ignore.case = TRUE)) |>
    left_join(nodes2 |> select(graph_id, chapter = graph_label), by = c("from_id" = "graph_id")) |>
    left_join(nodes2 |> select(graph_id, entity = graph_label, entity_type = graph_type), by = c("to_id" = "graph_id")) |>
    filter(!is.na(chapter), !is.na(entity), !grepl("^Ch\\.", entity)) |>
    arrange(desc(mentions), chapter, entity) |>
    select(chapter, entity, entity_type, mentions) |>
    head(20)

  if (nrow(top_links) > 0) {
    kable(top_links, caption = "Top chapter-to-entity relationships")
  } else {
    cat("No chapter-to-entity links found in the graph edge file.")
  }
} else {
  cat("Graph table could not be summarized because expected columns were not found.")
}
```
